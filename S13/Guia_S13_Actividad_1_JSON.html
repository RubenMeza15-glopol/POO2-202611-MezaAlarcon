2. Actividad 1 — RepositorioJSON

30 minutos

¿Qué hace?

Implementa un repositorio que guarda estudiantes en un archivo JSON. Al crearse, intenta leer el archivo existente. Cuando guardamos o eliminamos un estudiante, actualiza el archivo para que el cambio permanezca.

Archivo: repositorio_json.py — código comentado línea por línea

import json                         # Permite convertir datos Python a JSON y viceversa.
import pathlib                      # Facilita trabajar con rutas y archivos.
from dataclasses import asdict      # Convierte un objeto dataclass en un diccionario.
from dataclasses import fields      # Importado en la guía; en este código no se utiliza.
from typing import Optional         # Permite indicar que una búsqueda puede devolver None.
from estudiante import Estudiante   # Importa el modelo Estudiante del proyecto.

class RepositorioJSON:              # Declara el repositorio que almacena estudiantes en JSON.

    def __init__(self, ruta: str) -> None:  # Constructor: recibe la ubicación del archivo.
        self._ruta = pathlib.Path(ruta)    # Convierte la ruta de texto en un objeto Path.
        self._datos: dict[int, Estudiante] = {}  # Crea un diccionario: código -> estudiante.
        self._cargar()                     # Recupera los datos guardados al iniciar.

    def _cargar(self) -> None:       # Método interno que lee el archivo JSON.
        if not self._ruta.exists():  # Comprueba si el archivo ya existe.
            return                  # Si no existe, termina; el repositorio queda vacío.
        with self._ruta.open(encoding="utf-8") as f:  # Abre el archivo para lectura.
            for item in json.load(f): # Lee el JSON y recorre cada estudiante convertido en diccionario.
                e = Estudiante(**item) # Crea un Estudiante usando las claves del diccionario.
                self._datos[e.codigo] = e # Guarda el objeto en memoria usando su código como clave.

    def _persistir(self) -> None:    # Método interno que escribe los datos en el archivo.
        lista = [asdict(e) for e in self._datos.values()] # Convierte cada estudiante en diccionario.
        with self._ruta.open("w", encoding="utf-8") as f: # Abre el archivo en modo escritura.
            json.dump(lista, f, indent=2, # Escribe la lista en formato JSON legible.
                      ensure_ascii=False) # Conserva correctamente tildes y caracteres españoles.

    def guardar(self, e: Estudiante) -> None: # Inserta o reemplaza un estudiante.
        self._datos[e.codigo] = e    # Coloca el estudiante en el diccionario por su código.
        self._persistir()            # Guarda inmediatamente el estado actualizado en JSON.

    def buscar(self, cod: int) -> Optional[Estudiante]: # Busca por código.
        return self._datos.get(cod)  # Devuelve el estudiante o None si no existe.

    def listar(self) -> list[Estudiante]: # Devuelve todos los estudiantes.
        return list(self._datos.values()) # Convierte los valores del diccionario en una lista.

    def eliminar(self, cod: int) -> None: # Elimina al estudiante cuyo código se recibe.
        self._datos.pop(cod, None)   # Elimina la clave; si no existe, no produce error.
        self._persistir()            # Reescribe el JSON sin el estudiante eliminado.

Cómo entender el recorrido

  1. Se crea RepositorioJSON("estudiantes.json").
  2. El constructor prepara el diccionario y llama a _cargar().
  3. Si el archivo existe, sus registros se convierten nuevamente en objetos Estudiante.
  4. Al llamar a guardar(), el estudiante se añade al diccionario y se ejecuta _persistir().
  5. Al crear otra instancia apuntando al mismo archivo, se vuelven a cargar los datos. Eso demuestra la persistencia.

Archivo: test_json_persistencia.py — prueba comentada

import os                              # Permite eliminar el archivo temporal al terminar.
import pathlib                         # Se importa en la guía; no se usa en esta prueba.
from repositorio_json import RepositorioJSON # Importa el repositorio JSON.
from estudiante import Estudiante     # Importa el modelo de estudiante.

RUTA = "/tmp/estudiantes_test.json"   # Define el archivo temporal usado por la prueba.

def test_persistencia_entre_instancias(): # Define la prueba de persistencia.
    # Primera instancia: guardar estudiantes.
    r1 = RepositorioJSON(RUTA)        # Crea el primer repositorio y carga el archivo si existe.
    r1.guardar(Estudiante(1001, "Ana Quispe", 15.5, [14.0, 18.0, 14.5])) # Guarda a Ana.
    r1.guardar(Estudiante(1002, "Juan Mamani", 12.0, [10.0, 12.0, 14.0])) # Guarda a Juan.

    # Segunda instancia: comprobar que los datos quedaron en el archivo.
    r2 = RepositorioJSON(RUTA)        # Crea otro repositorio leyendo el mismo archivo.
    ana = r2.buscar(1001)             # Busca a Ana desde la segunda instancia.
    assert ana is not None            # Falla si Ana no fue recuperada.
    assert ana.nombre == "Ana Quispe" # Comprueba que el nombre sea correcto.
    assert abs(ana.ppa - 15.5) < 0.01 # Comprueba el PPA con una pequeña tolerancia.

    juan = r2.buscar(1002)            # Busca a Juan desde la segunda instancia.
    assert juan is not None            # Comprueba que Juan también se recuperó.
    assert len(r2.listar()) == 2      # Comprueba que hay exactamente dos estudiantes.

    os.remove(RUTA)                   # Elimina el archivo temporal para limpiar la prueba.

test_persistencia_entre_instancias()  # Ejecuta la función de prueba directamente.
print("RepositorioJSON: persistencia verificada ✓") # Muestra el mensaje si no hubo fallos.
Resultado esperado: RepositorioJSON: persistencia verificada ✓. La segunda instancia encontró a Ana y Juan y verificó que había dos registros. Si un assert falla, Python detiene la prueba con un error.

¿Qué demuestra el adicional eliminar(cod)?

Que no basta con borrar el registro del diccionario temporal: también se debe llamar a _persistir(). De esa manera, el archivo JSON queda actualizado y el estudiante no reaparece al reiniciar.
