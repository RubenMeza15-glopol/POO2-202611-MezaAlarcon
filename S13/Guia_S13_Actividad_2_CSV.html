3. Actividad 2 — RepositorioCSV

20 minutos

¿Qué hace?

Guarda los estudiantes en un archivo CSV. CSV es un formato de texto organizado en filas y columnas, que puede abrirse con VS Code o LibreOffice Calc. Como las notas son una lista, el código usa JSON para convertir esa lista en texto dentro de una celda.

Archivo: repositorio_csv.py — código comentado

import csv                           # Proporciona lectores y escritores de archivos CSV.
import pathlib                       # Permite manejar la ruta del archivo.
from typing import Optional          # Indica que buscar puede devolver un estudiante o None.
from estudiante import Estudiante    # Importa la clase del modelo.

CAMPOS = ["codigo", "nombre", "ppa", "notas"] # Define los nombres y el orden de las columnas.

class RepositorioCSV:                 # Declara el repositorio que utiliza un archivo CSV.

    def __init__(self, ruta: str) -> None: # Constructor que recibe la ruta del CSV.
        self._ruta = pathlib.Path(ruta)    # Guarda la ruta como objeto Path.
        self._datos: dict[int, Estudiante] = {} # Crea el diccionario interno de estudiantes.
        self._cargar()                     # Lee los datos existentes al iniciar.

    def _cargar(self) -> None:         # Lee los registros del archivo CSV.
        if not self._ruta.exists():    # Verifica si el archivo existe.
            return                    # Si no existe, termina con el diccionario vacío.
        with self._ruta.open(newline="", encoding="utf-8") as f: # Abre el CSV para lectura.
            for row in csv.DictReader(f): # Lee cada fila como diccionario usando los encabezados.
                import json           # Importa JSON para reconstruir la lista de notas.
                e = Estudiante(       # Crea un objeto Estudiante con los datos de la fila.
                    int(row["codigo"]), # Convierte el código de texto a entero.
                    row["nombre"],    # Recupera el nombre como texto.
                    float(row["ppa"]), # Convierte el PPA de texto a número decimal.
                    json.loads(row["notas"])) # Convierte el texto JSON de notas en una lista.
                self._datos[e.codigo] = e # Guarda el estudiante en el diccionario por código.

    def _persistir(self) -> None:      # Escribe el contenido actual del diccionario al CSV.
        import json                    # Permite convertir la lista de notas a texto JSON.
        with self._ruta.open("w", newline="", encoding="utf-8") as f: # Abre el CSV para escritura.
            w = csv.DictWriter(f, fieldnames=CAMPOS) # Crea el escritor con las columnas definidas.
            w.writeheader()             # Escribe la primera fila con los nombres de columnas.
            for e in self._datos.values(): # Recorre todos los estudiantes almacenados.
                w.writerow({"codigo": e.codigo, # Escribe el código en su columna.
                    "nombre": e.nombre,  # Escribe el nombre.
                    "ppa": e.ppa,        # Escribe el promedio.
                    "notas": json.dumps(e.notas)}) # Guarda la lista de notas como texto JSON.

    def guardar(self, e: Estudiante) -> None: # Guarda o reemplaza un estudiante.
        self._datos[e.codigo] = e     # Actualiza el diccionario usando el código.
        self._persistir()             # Actualiza el archivo CSV.

    def buscar(self, cod: int) -> Optional[Estudiante]: # Busca un estudiante por código.
        return self._datos.get(cod)   # Devuelve el objeto o None si no existe.

    def listar(self) -> list[Estudiante]: # Devuelve todos los estudiantes.
        return list(self._datos.values()) # Retorna una lista con los objetos almacenados.

¿Cómo se ve el archivo?

codigo,nombre,ppa,notas
1001,Ana Quispe,15.5,"[14.0, 18.0, 14.5]"
1002,Juan Mamani,12.0,"[10.0, 12.0, 14.0]"

La primera fila es el encabezado. Cada fila siguiente representa un estudiante. Observa que codigo y ppa se leen inicialmente como texto desde CSV y por eso se convierten con int() y float(). Las notas se reconstruyen con json.loads().
